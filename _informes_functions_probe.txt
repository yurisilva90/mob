async function infCarregarBadgesListagem(){
  if(!_SUPA) return;
  const els=[...document.querySelectorAll('.inf-vitem-badges[data-vname]')];
  if(!els.length) return;
  await Promise.all(els.map(async el=>{
    const nome=el.getAttribute('data-vname');
    try{
      const {data:vc}=await _SUPA.from('venue_cache').select('id,lat,lng').ilike('name',nome).limit(1);
      if(!vc||!vc.length) return;
      const venue=vc[0];
      const d=0.003;
      const [{data:stats},{data:posts}]=await Promise.all([
        _SUPA.rpc('venue_collab_stats',{p_venue_id:venue.id}),
        _SUPA.from('feed_posts').select('param_value,param_detail,created_at')
          .eq('type','dinamico').eq('status','ativo').gt('expires_at',new Date().toISOString())
          .gte('lat',venue.lat-d).lte('lat',venue.lat+d).gte('lng',venue.lng-d).lte('lng',venue.lng+d)
          .order('created_at',{ascending:false}).limit(10)
      ]);
      let html='';
      if(stats && stats.alta_demanda_agora){
        html+=`<div class="inf-badge-btn peak"><svg viewBox="0 0 24 24" width="14" height="14" fill="#fff" stroke="none"><path d="M12 2c1 3-2 4-2 7a4 4 0 0 0 8 0c0-1-.5-2-1-2 1 3-1 4-2 4a2 2 0 0 1-2-2c0-2 2-3 2-5-2 0-4 1-5 3-1 2-1 4 0 6a6 6 0 0 0 12-2c0-4-3-6-4-9-1 1-3 3-3 5-1-2-1-3-1-5-2 1-2 3-2 5z"/></svg><span>ALTA</span></div>`;
      }
      const dyn99=(posts||[]).find(p=>p.param_value==='99');
      if(dyn99){
        html+=`<div class="inf-badge-btn dyn"><svg viewBox="0 0 24 24" width="14" height="14" fill="#fff" stroke="none"><path d="M13 2 3 14h7l-1 8 10-12h-7z"/></svg><span>99: ${esc(dyn99.param_detail||'ativo')}</span></div>`;
      }
      el.innerHTML=html;
    }catch(e){ /* silencioso — badge é acessório, não pode travar a lista */ }
  }));
}

function infHaversineM(lat1,lon1,lat2,lon2){
  const R=6371000, toRad=x=>x*Math.PI/180;
  const dLat=toRad(lat2-lat1), dLon=toRad(lon2-lon1);
  const a=Math.sin(dLat/2)**2 + Math.cos(toRad(lat1))*Math.cos(toRad(lat2))*Math.sin(dLon/2)**2;
  return R*2*Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

async function infCarregarContagens(){

    // reformulação de 16/08/2026 — segue existindo só pro cron de
    // expiração antigo) pra feed_posts. feed_posts não tem location_name
    // (é geolocalizado, não por local nomeado), então a contagem vira
    // match textual no endereço geocodificado — aproximado, não é mais
    // igualdade exata como era com location_name. Efeito prático: alguns
    // locais podem subcontar se o endereço reverso não citar o nome
    // exatamente (ex.: aeroporto que geocodifica só como rua) — degrada
    // pra "–" nesse caso, nunca quebra a tela.
    const{data}=await _SUPA.from('feed_posts')
      .select('address')
      .eq('status','ativo')
      .gt('expires_at',new Date().toISOString())
      .not('address','is',null);
    if(!data)return;
    const countFor=(nome)=>{
      const needle=(nome||'').toLowerCase().trim();
      if(!needle) return 0;
      return data.filter(r=>(r.address||'').toLowerCase().includes(needle)).length;
    };

    const d=await r.json();
    if(!d.elements||!d.elements.length) return;
    // Calcula distância e ordena
    const R=6371;
    const dist=(a,b,c,dd)=>R*2*Math.asin(Math.sqrt(Math.sin(((c-a)*Math.PI/180)/2)**2+Math.cos(a*Math.PI/180)*Math.cos(c*Math.PI/180)*Math.sin(((dd-b)*Math.PI/180)/2)**2));
    const itens=d.elements
      .filter(e=>e.tags&&e.tags.name)
      .map(e=>({n:e.tags.name, sub:e.tags['addr:suburb']||e.tags['addr:city']||'Via GPS', km:dist(lat,lon,e.lat||lat,e.lon||lon)}))
      .sort((a,b)=>a.km-b.km)
      .slice(0,12)
      .map(e=>({...e,distStr:(e.km<1?Math.round(e.km*1000)+'m':e.km.toFixed(1)+'km')}));
    if(itens.length) infAbrirVenueSheet(cat, itens, true);
