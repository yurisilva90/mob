// Chamada pelo nativo ao voltar ao foreground — sincroniza GPS
function nativeSyncGps(km, startMs, pausedMs) {
  if (!S || !S.gps || S.gps.st === 'idle') return;

  // GUARD: nunca deixa o sync nativo DIMINUIR o km acumulado.
  // Se o GpsService foi morto e reiniciado pelo Android, ele pode chamar
  // nativeSyncGps com km < do que ja temos em memoria. Nesse caso mantemos
  // o valor maior (o que estava em S.gps.km ou no nativo, o que for maior).
  const kmNat = (typeof km === 'number' && isFinite(km)) ? km : S.gps.km;
  // GUARD DE PLAUSIBILIDADE: se o serviço nativo Android nunca for
  // efetivamente resetado entre uma jornada e outra (não existe um comando
  // nativo pra isso hoje), ele pode reportar o km ACUMULADO de uma jornada
  // ANTERIOR logo no início de uma jornada NOVA — e como o guard acima
  // nunca deixa diminuir, esse km 'órfão' seria herdado silenciosamente.
  // Rejeita qualquer salto que implique velocidade fisicamente impossível
  // (>180km/h) dado o tempo decorrido desde o início desta sessão.
  const elapsedMin = S.gps.start ? (Date.now() - S.gps.start) / 60000 : 0;
  const maxPlausibleKm = (elapsedMin * 3) + 0.5; // 180km/h + margem
  if (kmNat > maxPlausibleKm) {
    console.warn('[nativeSyncGps] Ignorado — km implausível pro tempo decorrido:', kmNat.toFixed(1), 'km em', elapsedMin.toFixed(1), 'min (provável km órfão do serviço nativo)');
  } else if (kmNat > (S.gps.km || 0)) {
    S.gps.km = kmNat;
  }
  if (startMs && !S.gps.start) S.gps.start = startMs;
  // GUARD: nunca deixa o sync nativo DIMINUIR o pausedMs acumulado.
  // Se o GpsService reiniciou com pausedMs=0, mantemos o valor JS que é maior.
  const pauseNat = (typeof pausedMs === 'number' && isFinite(pausedMs)) ? pausedMs : 0;
  if (pauseNat >= (S.gps.pausedMs || 0)) { S.gps.pausedMs = pauseNat; }

  // LIMPA pontos de referencia JS para evitar dupla contagem.
  // Quando o app volta ao foreground, restoreSession recarrega o ultimo ponto
  // GPS salvo (g.pts = [lastPt]). Se o carro se deslocou no background, o
  // proximo disparo do watchPosition calcularia essa distancia E SOMARIA em
  // cima do km ja atualizado pelo nativo -> inflaria o total momentaneamente.
  // Zerando g.pts e lastFixTime, o proximo fix apenas registra a posicao atual
  // como nova referencia sem contar nenhuma distancia extra.
  S.gps.pts = [];
  S.gps.lastFixTime = null;

  const kmFinal = S.gps.km;
  // Só atualiza DOM se o usuário estiver vendo hoje
  if (S.curDate === DB.today()) {
    setText('jc-km', kmFinal.toFixed(1));
    const dayNow = DB.day(DB.today());
    const finKm = (dayNow.sessions||[]).reduce((a,x)=>a+(x.km||0),0);
    setText('jc-kmtotal', (finKm + kmFinal + (dayNow.kmAdjust||0)).toFixed(1)+' km');
  }

  // Recria a bolinha flutuante se o Android tiver matado o overlay.
  // startFloating e seguro de chamar de novo: se ja existe, so atualiza o km.
  if (window.SmartMobiNative && DB.cfg().overlayEnabled !== false) {
    try {
      if (typeof SmartMobiNative.startFloating === 'function') SmartMobiNative.startFloating(S.gps.start, kmFinal);
      if (typeof SmartMobiNative.updateFloatingStatus === 'function') SmartMobiNative.updateFloatingStatus('running');
    } catch(e) {}
  }
}

function editJss(dateStr, idx) {
  const day = DB.day(dateStr);
  const sess = (day.sessions || [])[idx];
  if (!sess) return;

  document.getElementById('jss-edit-date').value = dateStr;
  document.getElementById('jss-edit-idx').value  = idx;
  document.getElementById('jss-edit-km').value   = (sess.km || 0).toFixed(1);

  // Hora início e fim a partir dos timestamps da sessão
  const fmt = ms => {
    if (!ms) return '';
    const d = new Date(ms);
    return String(d.getHours()).padStart(2,'0') + ':' + String(d.getMinutes()).padStart(2,'0');
  };
  document.getElementById('jss-edit-inicio').value = fmt(sess.start);
  document.getElementById('jss-edit-fim').value    = fmt(sess.end);

  openMo('mo-jss-edit');
}

function saveJss() {
  const dateStr = document.getElementById('jss-edit-date').value;
  const idx     = parseInt(document.getElementById('jss-edit-idx').value);
  const km      = parseFloat(document.getElementById('jss-edit-km').value) || 0;
  const inicioStr = document.getElementById('jss-edit-inicio').value; // "HH:MM"
  const fimStr    = document.getElementById('jss-edit-fim').value;    // "HH:MM"

  const day = DB.day(dateStr);
  if (!day.sessions || !day.sessions[idx]) { closeMo('mo-jss-edit'); return; }

  const sess = day.sessions[idx];

  // Recalcular timestamps de início e fim mantendo a data original
  const toMs = (base, timeStr) => {
    if (!timeStr) return base;
    const [hh, mm] = timeStr.split(':').map(Number);
    const d = new Date(base || Date.now());
    d.setHours(hh, mm, 0, 0);
    return d.getTime();
  };
  const newStart = toMs(sess.start, inicioStr);
  const newEnd   = toMs(sess.end || sess.start, fimStr);
  // Elapsed recalculado a partir da diferença início→fim
  const elapsed  = newEnd > newStart ? Math.round((newEnd - newStart) / 1000) : (sess.elapsed || 0);

  sess.km      = km;
  sess.start   = newStart;
  sess.end     = newEnd;
  sess.elapsed = elapsed;
  DB.saveDay(day);

  // Sincronizar com Supabase
  if (_supaUser && sess.id) {
    _SUPA.from('sessions').update({
      km:           parseFloat(km.toFixed(3)),
      start_ms:     newStart,
      end_ms:       newEnd,
      elapsed_secs: elapsed
    }).eq('id', sess.id).then(() => {}).catch(() => {});
  }

  closeMo('mo-jss-edit');
  renderJornada();
  if (S.sec === 'relatorios') renderHome();
  showToast('Sessão atualizada');
}

function deleteJss(dateStr, idx) {
  showConfirm('Excluir esta sessão?', () => {
    const day = DB.day(dateStr);
    if (!day.sessions || !day.sessions[idx]) return;

    const sess = day.sessions[idx];
    day.sessions.splice(idx, 1);
    DB.saveDay(day);

    // Remover do Supabase (delete real, não update)
    if (sess.id) {
      SYNC.del('sessions', { id: sess.id });
    }

    renderJornada();
    if (S.sec === 'relatorios') renderHome();
    showToast('Sessão excluída');
  }, { confirmLabel: 'Excluir', danger: true, cancelLabel: 'Cancelar' });
}

function sizeRouteLine(panelEl) {
  try {
    const block = panelEl.querySelector('.route-block');
    if (!block) return;
    const marks = block.querySelectorAll('.route-mark');
    const line = block.querySelector('.route-line');
    if (!line || marks.length < 2) return;
    const first = marks[0], last = marks[marks.length - 1];
    const blockTop = block.getBoundingClientRect().top;
    const firstMid = first.getBoundingClientRect().top - blockTop + first.offsetHeight / 2;
    const lastMid = last.getBoundingClientRect().top - blockTop + last.offsetHeight / 2;
    if (lastMid <= firstMid) return;
    line.style.top = firstMid + 'px';
    line.style.height = (lastMid - firstMid) + 'px';
  } catch (e) {}
}

async function authInit() {
  ['app-nav','app-header'].forEach(id => { const el=document.getElementById(id);if(el)el.style.display='none'; });
  document.querySelectorAll('.screen').forEach(s => s.style.display='none');
  await new Promise(r => setTimeout(r, 300));
  const { data:{ session } } = await _SUPA.auth.getSession();
  if (session?.user) { _supaUser=session.user; await loadProfile(); showApp(); }
  else showLogin();
  _SUPA.auth.onAuthStateChange(async (ev, session) => {
    if ((ev==='SIGNED_IN' || ev==='TOKEN_REFRESHED') && session?.user) {
      _supaUser = session.user;
      // Salva tokens no storage nativo (persiste mesmo com app morto)
      if (window.SmartMobiNative?.saveSession) {
        try { SmartMobiNative.saveSession(JSON.stringify({
          access_token: session.access_token,
          refresh_token: session.refresh_token
        })); } catch(e) {}
      }
      // Salva credenciais para o GpsService usar nas notificações de reporte
      // rápido e no alerta automático de dinâmico do feed. Reenviado também
      // a cada TOKEN_REFRESHED (autoRefreshToken troca o token ~1x/hora) —
      // antes só rodava no SIGNED_IN, então esses recursos paravam de
      // funcionar (silenciosamente) depois de ~1h de app aberto.
      if (window.SmartMobiNative?.saveUserToken) {
        try { SmartMobiNative.saveUserToken(session.user.id, session.access_token); } catch(e) {}
      }
      if (ev==='SIGNED_IN') { await loadProfile(); showApp(); }
    } else if (ev==='SIGNED_OUT') {
      _supaUser = null;
      if (window.SmartMobiNative?.saveSession) {
        try { SmartMobiNative.saveSession(''); } catch(e) {}
      }
      showLogin();
    }
  });
}

  if (!_appBooted) {
    _appBooted = true;
    // FIX: restaura a última tela em vez de forçar Home quando a WebView
    // recarrega (Android mata a página em background)
    let lastSec = 'home';
    try {
      const saved = JSON.parse(localStorage.getItem('sm_last_screen') || 'null');
      // Só restaura se foi salvo há menos de 30 min (senão, home mesmo)
      if (saved && saved.sec && (Date.now() - saved.ts) < 30*60*1000) lastSec = saved.sec;
    } catch(e) {}
    navTo(lastSec);
    _bootPullBg();
  }
}

// Promessa que resolve quando a restauração da nuvem (pullAll) termina no
// boot. Telas como Custo do Veículo e MōB Flash usam isso pra não mostrar
// formulário vazio e deixar o usuário preencher de novo por cima de dados
// que já estavam guardados, só porque a restauração ainda não tinha chegado.
let _bootPullResolve;
let _bootPullDone = false;
const _bootPullPromise = new Promise(r => { _bootPullResolve = r; });

// Depois de _bootPullPromise pronta: espera ela terminar (com timeout, pra
// não travar pra sempre se estiver offline) antes de liberar uma tela.
async function _waitBootPull(maxMs = 4000) {
  try { await Promise.race([_bootPullPromise, new Promise(r => setTimeout(r, maxMs))]); } catch (e) {}
}

// Pull do cloud em BACKGROUND — a UI já está visível com dados locais.
// Quando terminar, re-renderiza tudo e restaura sessão GPS do cloud se houver.
async function _bootPullBg() {
  try { await SYNC.pullAll(); } catch(e) { console.error('pullAll boot:', e); }
  try { await SYNC.flushQueue(); } catch(e) { console.error('flushQueue boot:', e); }
  try { checkRecurringLancamentos(); } catch(e) { console.error('checkRecurringLancamentos boot:', e); }
  try{restoreGpsSession();} catch(e){}
  try{syncNavButton();}  catch(e){}
  try{renderHome();}     catch(e){}
  try{renderJornada();}  catch(e){}
  try{renderFin();}      catch(e){}
  try{renderAgenda();} catch(e){}
  try{ if (S.sec==='corridas') renderCorridasActive(); } catch(e){}
  try{ if (S.sec==='lancamentos') renderLancamentos(); } catch(e){}
  _bootPullDone = true;
  _bootPullResolve();
}
