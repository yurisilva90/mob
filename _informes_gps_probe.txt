    turistico:`node(around:3000,${lat},${lon})[tourism~"^(museum|attraction|viewpoint|gallery)$"][name];`,

    const d=await r.json();
    if(!d.elements||!d.elements.length) return;
    // Calcula distância e ordena
    const R=6371;

    const itens=d.elements
      .filter(e=>e.tags&&e.tags.name)

      .sort((a,b)=>a.km-b.km)
      .slice(0,12)
      .map(e=>({...e,distStr:(e.km<1?Math.round(e.km*1000)+'m':e.km.toFixed(1)+'km')}));
    if(itens.length) infAbrirVenueSheet(cat, itens, true);
  }catch(e){console.warn('overpass venues',e);}
}
