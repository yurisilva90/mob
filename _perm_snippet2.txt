function flashToggleKpi(key, el) {

    const enabledCount = Object.values(flashDraft.kpis).filter(k => k.enabled).length;

  renderFlashRecusaList();
  renderFlashPreview();
}

let flashRecusaNumKey = null;
function flashOpenRecusaNum(key) {
  flashRecusaNumKey = key;
  const d = FLASH_RECUSA_DEFS.find(x => x.key === key);
  const r = flashDraft.recusas[key];
  document.getElementById('flash-recusa-num-ico').textContent = d.icon;
  document.getElementById('flash-recusa-num-title').textContent = d.label;
  document.getElementById('flash-recusa-num-phrase').textContent = d.phrase(r.value);
  document.getElementById('flash-recusa-num-unit').textContent = d.unit || '';
  const slider = document.getElementById('flash-recusa-num-slider');
  slider.min = d.min; slider.max = d.max; slider.step = d.step; slider.value = r.value;
  _flashRecusaNumRender();
  document.getElementById('flash-step1').style.display = 'none';
  document.getElementById('flash-step2').style.display = 'none';
  document.getElementById('flash-ruler').style.display = 'none';
  document.getElementById('flash-enderecos').style.display = 'none';
  document.getElementById('flash-recusa-num').style.display = '';
}
function _flashRecusaNumRender() {
  const d = FLASH_RECUSA_DEFS.find(x => x.key === flashRecusaNumKey);
  const r = flashDraft.recusas[flashRecusaNumKey];
  document.getElementById('flash-recusa-num-val').textContent = flashFmtNum(r.value);
  document.getElementById('flash-recusa-num-phrase').textContent = d.phrase(r.value);
  document.getElementById('flash-recusa-num-slider').value = r.value;
}
function flashRecusaNumStep(dir) {
  const d = FLASH_RECUSA_DEFS.find(x => x.key === flashRecusaNumKey);
  const r = flashDraft.recusas[flashRecusaNumKey];
  let v = Math.round((r.value + dir * d.step) / d.step) * d.step;
  r.value = Math.min(d.max, Math.max(d.min, Math.round(v * 100) / 100));
  _flashRecusaNumRender();
}
function flashRecusaNumFromSlider(val) {
  const r = flashDraft.recusas[flashRecusaNumKey];
  r.value = parseFloat(val);
  _flashRecusaNumRender();
}
function flashRecusaNumClose() {
  flashRecusaNumKey = null;
  _flashWizardShowStep(1);
  renderFlashRecusaList();
}
