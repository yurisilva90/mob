// Corrige o caso em que uma corrida foi apagada direto no Supabase (ex:

async function resyncCorridasFromCloud() {
  if (!_supaUser) { showToast('Sem conexão com a nuvem.'); return; }
  showConfirm('Recarregar da nuvem agora?\nCorridas confirmadas sincronizadas antes que sumiram do servidor saem do celular. Nada que ainda esteja pendente de subir é tocado.', async () => {
    try {
      await SYNC.pullAll();
      renderCorridasActive();
      if (S.sec === 'jornada') renderJornada();
      if (S.sec === 'relatorios') renderHome();
      /* sincronização silenciosa: sem toast de sucesso */
    } catch (e) {
      console.error('resyncCorridasFromCloud', e);
      showToast('Erro ao recarregar da nuvem.');
    }

function confirmDeleteTrip(date, id) {
  const day = DB.day(date);
  const t = (day.trips||[]).find(x => x.id === id);
  if (!t) return;
  const dest = t.dest ? ' → '+esc(t.dest) : '';
  showConfirm(`Excluir essa corrida?\n${DB.fR(t.value)} · ${t.origin?esc(t.origin):'—'}${dest}`, () => {
    day.trips = day.trips.filter(x => x.id !== id);
    DB.saveDay(day);
    SYNC.del('trips', { id });
    renderCorridasActive();
    if (date === S.curDate) { renderJornada(); }
    showToast('Corrida excluída.');
  }, { confirmLabel: 'Excluir' });
}

async function fetchBestOfferTexts(t) {
  // Retorna { texts, reason } em vez de só null/array — pedido do Yuri
  // (24/07/2026): precisa ver o motivo exato na tela (sem abrir console do
  // navegador) quando não achar nada, pra investigar mais rápido.
  if (!_supaUser) return { texts: null, reason: 'sem sessão ativa' };
  const plat = t.platform === 'uber' ? 'UBER' : '99';
  const anchor = t.accepted_at || t.trip_started_at;
  if (!anchor) return { texts: null, reason: 'corrida sem horário de aceite salvo' };
  const anchorMs = new Date(anchor).getTime();
  const from = new Date(anchorMs - 10*60*1000).toISOString();
  const to = new Date(anchorMs + 30*1000).toISOString();
  const { data, error } = await _SUPA.from('trip_reader_log').select('texts,created_at')
    .eq('user_id', _supaUser.id).eq('platform', plat)
    .gte('created_at', from).lte('created_at', to)
    .order('created_at', { ascending: false })
    // CONFIRMADO EM DADO REAL (24/07/2026): um intervalo de 10min30s desses
    // teve 523 linhas no total (todo tipo de leitura, não só oferta) — o
    // limite de 500 cortava fora justamente as leituras mais antigas, que
    // eram as que tinham a oferta. Margem bem folgada agora.
    .limit(3000);
  if (error) {
    console.error('fetchBestOfferTexts', error);
    return { texts: null, reason: 'erro na busca: '+(error.message || error.code || '?') };
  }
  if (!data || !data.length) return { texts: null, reason: `0 linhas no log entre ${from.slice(11,19)} e ${to.slice(11,19)} (plataforma ${plat})` };
  let best = null, bestLen = -1, offerRows = 0;
  data.forEach(row => {
    const st = row.texts && row.texts.state;
    if (st !== 'OFERTA' && st !== 'OFERTA_OCR') return;
    offerRows++;
    const raw = row.texts && row.texts.raw;
    if (Array.isArray(raw) && raw.length > bestLen) { best = raw; bestLen = raw.length; }
  });
  if (!best) return { texts: null, reason: `${data.length} linhas no log, nenhuma de oferta (plataforma ${plat})` };

async function _computeAutoTripRefresh(tripId) {
  const { data: rows, error: e1 } = await _SUPA.from('auto_trips').select('*').eq('id', tripId).limit(1);
  if (e1 || !rows || !rows.length) return { ok:false, reason:'corrida não encontrada' };
  const t = rows[0];
  const { texts, reason } = await fetchBestOfferTexts(t);
  if (!texts) return { ok:false, reason };
  const parsed = parseOfferJS(texts);
  if (parsed.valor == null) return { ok:false, reason:'texto insuficiente pra reconferir' };
  // Só os campos NUMÉRICOS — testado contra casos reais, o valor sai
  // confiável em várias leituras diferentes da mesma oferta, mas o
  // endereço às vezes sai lixo ("% LUCRO" etc) mesmo na leitura mais
  // completa. Melhor não arriscar sobrescrever um endereço bom.
  const patch = { offer_value: parsed.valor, status: 'confirmada' };
  if (parsed.dinamico != null) patch.offer_dinamico = parsed.dinamico;
  if (parsed.kmPickup != null) patch.offer_km_pickup = parsed.kmPickup;
  if (parsed.kmTrip != null) patch.offer_km_trip = parsed.kmTrip;
  if (parsed.minPickup != null) patch.offer_duration_pickup_sec = Math.round(parsed.minPickup*60);
  if (parsed.minTrip != null) patch.offer_duration_trip_sec = Math.round(parsed.minTrip*60);
  // Compara ANTES de salvar — pedido do Yuri (24/07/2026): mensagem
  // precisa dizer o que de fato mudou, não só "reconferida" genérico.
  const numEq = (a,b,tol) => (a==null && b==null) || (a!=null && b!=null && Math.abs(a-b) <= (tol||0.005));
  const changes = [];
  if (!numEq(t.offer_value, patch.offer_value, 0.005)) changes.push('valor '+DB.fR(patch.offer_value));
  if (patch.offer_dinamico != null && !numEq(t.offer_dinamico, patch.offer_dinamico, 0.005)) changes.push('dinâmico '+DB.fR(patch.offer_dinamico));
  if (patch.offer_km_trip != null && !numEq(t.offer_km_trip, patch.offer_km_trip, 0.05)) changes.push('km previsto');
  if (patch.offer_duration_trip_sec != null && !numEq(t.offer_duration_trip_sec, patch.offer_duration_trip_sec, 30)) changes.push('tempo previsto');
  // Diferença grande de valor (>15%, ou >R$5 pra valores pequenos) — pede
  // confirmação em vez de aplicar direto. Pedido do Yuri (16/08/2026):
  // caso real onde o valor pulou de R$42 pra R$26,20 sem aviso nenhum —
  // pode ser a correção certa (leitura mais completa) ou pode ser a nova
  // leitura que tá errada; sem ver as duas, não dá pra saber qual confiar.
  const oldVal = t.offer_value != null ? Number(t.offer_value) : null;
  const newVal = patch.offer_value;
  let valueBigChange = false;
  if (oldVal != null && oldVal > 0) {
    const diff = Math.abs(newVal - oldVal);
    valueBigChange = diff > 5 && (diff / oldVal) > 0.15;
  }
  return { ok:true, tripId, patch, changes, oldVal, newVal, valueBigChange };
}

async function refreshAutoTripFromLog(tripId, btnEl) {
  if (btnEl) btnEl.classList.add('spinning');
  try {
    const r = await _computeAutoTripRefresh(tripId);
    if (!r.ok) { showToast('Não achei: '+r.reason, 'error'); return; }
    if (!r.changes.length) { showToast('Corrida atualizada com sucesso', 'success'); return; }
    const finish = async () => {
      const saved = await _applyAutoTripRefresh(r.tripId, r.patch);
      if (!saved) { showToast('Erro ao salvar.', 'error'); return; }
      showToast('Corrigido: '+r.changes.join(', '), 'success');
      S.autoTripsDate = null; S.autoTripsHomeRange = null;
      if (S.sec === 'jornada') renderJornada();
      if (S.sec === 'relatorios') renderHomeGraphs();
      if (S.sec === 'corridas') renderCorridasActive();
    };
    if (r.valueBigChange) {
      showConfirm(
        `O valor dessa corrida vai mudar de ${DB.fR(r.oldVal)} para ${DB.fR(r.newVal)} — uma diferença grande.\n\nConfirma que ${DB.fR(r.newVal)} é o valor certo?`,
        finish,
        { danger: false, confirmLabel: 'Sim, corrigir', cancelLabel: 'Cancelar' }
      );
    } else {
      await finish();
    }
  } catch (e) {
    console.error('refreshAutoTripFromLog', e);
    showToast('Erro ao reconferir.', 'error');
  } finally {
    if (btnEl) btnEl.classList.remove('spinning');
  }
}

// Detalhamento Online/Buscando passageiro/Em corrida — puxado direto da
// máquina de estados do APK (accepted_at/pickup_started_at/trip_started_at/
// trip_ended_at, gravados em auto_trips a cada transição real de tela).
// BUSCANDO = trip_started_at - pickup_started_at (deslocamento até o
// passageiro); EM CORRIDA = trip_ended_at - trip_started_at; ONLINE = todo
// o resto do tempo da sessão (esperando oferta / avaliando oferta). Só usa
// corridas automáticas (autoTrips) — corridas importadas manualmente não
// têm esses timestamps porque não vieram da leitura de tela em tempo real.
function computeStatusBreakdown(elapsedSec, kmTotal, autoTrips) {
  let corridaSec = 0, buscandoSec = 0, corridaKm = 0, buscandoKm = 0;
  (autoTrips || []).forEach(t => {
    if (t.trip_started_at && t.trip_ended_at) {
      const s = (new Date(t.trip_ended_at) - new Date(t.trip_started_at)) / 1000;
      if (s > 0) corridaSec += s;
      corridaKm += parseFloat(t.real_km_trip ?? t.offer_km_trip) || 0;
    }
    if (t.pickup_started_at && t.trip_started_at) {
      const s = (new Date(t.trip_started_at) - new Date(t.pickup_started_at)) / 1000;
      if (s > 0) buscandoSec += s;
      buscandoKm += parseFloat(t.real_km_pickup ?? t.offer_km_pickup) || 0;
    }
  });
  const total = Math.max(0, elapsedSec || 0);
  const onlineSec = Math.max(0, total - corridaSec - buscandoSec);
  const onlineKm = Math.max(0, (kmTotal || 0) - corridaKm - buscandoKm);
  const pct = s => total > 0 ? Math.round(s / total * 100) : 0;
  return {
    total,
    online:   { sec: onlineSec,   km: onlineKm,   pct: pct(onlineSec) },
    buscando: { sec: buscandoSec, km: buscandoKm, pct: pct(buscandoSec) },
    corrida:  { sec: corridaSec,  km: corridaKm,  pct: pct(corridaSec) },
  };
}

function uzRound2(n) { return Math.round(n * 100) / 100; }

// Aceita "10.50" e "10,50" — exportações da Uber variam o separador por região/locale.
function uzNum(raw) {
  if (raw == null || raw === '') return 0;
  return parseFloat(String(raw).replace(',', '.')) || 0;
}

function uzParseTs(raw) {
  if (!raw) return null;
  // O timestamp do zip vem em UTC (tem 'Z' no final) — converte pra Brasília (UTC-3)
  const raw2 = String(raw).trim();
  try {
    const d = new Date(raw2.includes('T') ? raw2 : raw2.replace(' ', 'T') + 'Z');
    if (isNaN(d.getTime())) return null;
    // subtrai 3h pra Brasília
    const brt = new Date(d.getTime() - 3 * 60 * 60 * 1000);
    const pad = n => String(n).padStart(2, '0');
    const dateStr = brt.getUTCFullYear() + '-' + pad(brt.getUTCMonth()+1) + '-' + pad(brt.getUTCDate());
    const timeStr = pad(brt.getUTCHours()) + ':' + pad(brt.getUTCMinutes());
    return { dateStr, timeStr };
  } catch(e) { return null; }
}

// Soma gorjeta e incentivo/promoção do período inteiro (informativo, não por corrida).
function uzAggregatePayments(rows) {
  let tip = 0, promo = 0;
  for (const r of rows) {
    const amt = uzNum(r['Local Amount']);
    if (r['Category'] === 'tip') tip += amt;
    if (r['Category'] === 'existing_driver_incentive') promo += amt;
  }
  return { tip: uzRound2(tip), promo: uzRound2(promo) };
}

// Monta as corridas concluídas direto do driver_lifetime_trips. Corridas canceladas
// (status diferente de 'completed') são ignoradas — não são corridas de fato.
function uzBuildTrips(rows) {
  const trips = [];
  for (const r of rows) {
    if (r['status'] !== 'completed') continue;
    const begin = uzParseTs(r['begintrip_timestamp_local']) || uzParseTs(r['request_timestamp_local']);
    if (!begin) continue;
    const value = uzNum(r['driver_upfront_fare_local']) || uzNum(r['original_fare_local']);
    trips.push({
      date: begin.dateStr, time: begin.timeStr,
      value: uzRound2(Math.max(0, value)),
      km: r['trip_distance_miles'] ? uzRound2(uzNum(r['trip_distance_miles']) * 1.60934) : 0,
      // o app guarda duração em MINUTOS (a IA de import por foto também usa minutos);
      // o export da Uber vem em segundos, por isso a divisão por 60 aqui.
      duration: r['trip_duration_seconds'] ? Math.round(uzNum(r['trip_duration_seconds']) / 60) : 0,
      category: r['global_product_name'] || '',
      surge: r['is_surged'] === 'true',
      airport: r['is_airport_trip'] === 'true',
      // dinamico junta tarifa dinamica (surge) + promocao/desconto da corrida num
      // valor só — já incluso no "value", mostrado separado só pra detalhar a corrida.
      // Pode ficar negativo se o desconto for maior que o surge.
      dinamico:    uzRound2(Math.max(0, uzNum(r['surge_fare_local'])) + uzNum(r['promotion_local'])),
      dinheiro:   r['is_cash_trip'] === 'true',
      pedagio:    uzNum(r['toll_amount_local']) > 0,
      serviceFee: (r['service_fee'] !== undefined && r['service_fee'] !== '') ? parseFloat(r['service_fee']) : null,
      taxaZero:   r['service_fee'] === '0' || r['service_fee'] === '0.0',
    });
  }
  return trips;
}

function uzFindEntry(zip, suffix) {
  const f = Object.values(zip.files).find(x => !x.dir && x.name.endsWith(suffix));
  if (!f) throw new Error('Não encontrei "' + suffix + '" no zip. Confira se é o arquivo certo da Uber.');
  return f;
}

// Tenta de novo (até 3x) antes de desistir de um print/frame — evita que um
// erro passageiro de rede derrube a corrida inteira do import silenciosamente.
async function extractWithRetry(fn, onRetry, maxAttempts) {
  maxAttempts = maxAttempts || 3;
  let lastErr;
  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try { return await fn(); }
    catch(e) {
      lastErr = e;
      if (attempt < maxAttempts) { onRetry(attempt+1, maxAttempts); await new Promise(r => setTimeout(r, 800)); }
    }
  }
  throw lastErr;
}
