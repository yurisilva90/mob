function loadAutoTripsForDay(dateStr) {
  if (!_supaUser) return [];
  // O dia da Jornada é o dia LOCAL do motorista. Converte meia-noite local
  // para UTC antes de consultar timestamptz — evita perder corridas após 21h
  // no Brasil quando o relógio UTC já virou para o dia seguinte.
  const fromLocal = new Date(dateStr+'T00:00:00');
  const toLocal = new Date(fromLocal); toLocal.setDate(toLocal.getDate()+1);
  const { data, error } = await _SUPA.from('auto_trips').select('*')
    .eq('user_id', _supaUser.id)
    .gte('trip_started_at', fromLocal.toISOString()).lt('trip_started_at', toLocal.toISOString())
    .order('trip_started_at', {ascending:true});
  // null = falha de rede/consulta; [] = consulta válida sem corridas.
  return error ? null : (data||[]);
}

function confirmImport() {
  const selected = (window._pendingTrips||[]).filter(t=>t._sel);
  if (!selected.length) { closeMo('mo-import-confirm'); return; }
  const day = DB.day(S.curDate);
  let added=0, skipped=0, updated=0;
  const toUpdateAuto = [];
  const toPushAuto = [];
  const videoId = (window._pendingImportVideoIds||[])[0] || null;
  selected.forEach(t => {
    // Já casada com uma auto_trip existente (matching já feito em
    // processImportBatch, antes da tela abrir) — só atualiza o valor,
    // nunca o endereço (esse já veio validado pela captura ao vivo/GPS).
    if (t._matchedAutoTripId) {
      toUpdateAuto.push({ id: t._matchedAutoTripId, value: parseFloat(t.value)||0, km: parseFloat(t.km)||0,
        dinheiro: !!t.dinheiro });
      updated++;
      return;
    }
    // Nova — nunca vista na captura ao vivo. Vira auto_trips já Confirmada
    // pelo vídeo; 'Importada' agora é somente a tag de origem — unifica com o resto dos
    // dados em vez de virar corrida "manual" separada. Endereço já
    // passou pela validação looksLikeAddress lá em processImportBatch
    // (se não passou, já veio vazio aqui — nunca inventa).
    let startMs;
    try {
      const opDate = window._pendingImportOperationalDate || S.curDate || DB.today();
      startMs = operationalTimestampForTime(t.platform || window._pendingImportPlatform, opDate, t.time||'00:00');
    } catch(e) { startMs = NaN; }
    if (!startMs || isNaN(startMs)) startMs = Date.now();
    const endMs = t.duration > 0 ? startMs + t.duration*60000 : null;
    toPushAuto.push({
      user_id: _supaUser.id,
      platform: t.platform === '99' ? '99' : 'uber',
      accepted_at: new Date(startMs).toISOString(),
      trip_started_at: new Date(startMs).toISOString(),
      trip_ended_at: endMs ? new Date(endMs).toISOString() : null,
      offer_value: parseFloat(t.value)||0,
      real_km_trip: parseFloat(t.km)||0,
      origin_address: t.originFull || null,
      dest_address: t.destFull || null,
      dinheiro: !!t.dinheiro,
      status: 'confirmada',
      capture_source: 'importada',
      value_needs_review: false,
      data_quality_flag: null,
      import_video_id: videoId
    });
    added++;
  });
  // Aplica as atualizações nas automáticas casadas — valor real por cima do
  // estimado, km real quando a importação trouxe (senão mantém o que já
  // tinha), status vira confirmada (agora tem valor confirmado de verdade).
  toUpdateAuto.forEach(u => {
    _SUPA.from('auto_trips').update({
      offer_value: u.value,
      real_km_trip: u.km > 0 ? u.km : undefined,
      dinheiro: u.dinheiro,
      status: 'confirmada',
      value_needs_review: false,
      data_quality_flag: null,
      import_video_id: videoId || undefined
    }).eq('id', u.id).then(() => {}).catch(() => {});
  });
  // Corridas novas — insere de uma vez em auto_trips.
  if (toPushAuto.length) {
    _SUPA.from('auto_trips').insert(toPushAuto).then(() => {}).catch((e) => console.warn('insert importada falhou', e));
  }
  // Atualiza a contagem de corridas confirmadas/criadas por esse vídeo —
  // só informativo, útil pra revisão manual depois.
  if (videoId) {
    _SUPA.from('import_videos').update({ trip_count: toUpdateAuto.length + toPushAuto.length }).eq('id', videoId).then(()=>{}).catch(()=>{});
  }
  window._pendingImportVideoIds = null;
  if (toUpdateAuto.length || toPushAuto.length) { S.autoTripsDate = null; } // força recarregar no próximo renderJornada
  DB.saveDay(day); closeMo('mo-import-confirm'); window._pendingTrips=null;

  const dateHint = S.curDate !== DB.today()
    ? ` (data: ${S.curDate.split('-').reverse().slice(0,2).join('/')})` : '';
  const parts = [];
  if (added > 0) parts.push(`${added} corrida${added!==1?'s':''} nova${added!==1?'s':''} importada${added!==1?'s':''}${dateHint}`);
  if (updated > 0) parts.push(`${updated} já capturada${updated!==1?'s':''} automaticamente, valor confirmado`);
  if (skipped > 0) parts.push(`${skipped} ignorada${skipped!==1?'s':''} (já existente${skipped!==1?'s':''})`);
  const baseMsg = parts.length ? parts.join('. ')+'.' : 'Nenhuma corrida nova.';

  try{renderJornada();}catch(e){}
  try{renderHome();}catch(e){}
  // Garantir que Corridas mostra a data certa independente do filtro
  if (S.curDate !== DB.today()) S.corridasPeriod = 'tudo';
  try{renderCorridasActive();}catch(e){}
  showToast(baseMsg, skipped > 0 ? 'info' : 'success');
}

function renderJornada() {
  if (!S || !DB) return;
  try { updateCustoBanner(); } catch(e) {}
  if (!S.curDate) S.curDate = DB.today();
  const today = DB.today(), d = S.curDate, isToday = d === today;
  const dt    = new Date(d+'T12:00');
  const dows  = ['Domingo','Segunda-feira','Terça-feira','Quarta-feira','Quinta-feira','Sexta-feira','Sábado'];
  const ms    = ['jan','fev','mar','abr','mai','jun','jul','ago','set','out','nov','dez'];
  const msLong= ['janeiro','fevereiro','março','abril','maio','junho','julho','agosto','setembro','outubro','novembro','dezembro'];
  const dateEl = document.getElementById('dn-date');
  // Formato compacto: "23 jun 2026" — cabe em 1 linha
  if (dateEl) dateEl.textContent = isToday ? 'Hoje' : (dt.getDate()+' de '+msLong[dt.getMonth()]+' de '+dt.getFullYear());
  const dowEl = document.getElementById('dn-dow');
  // Dia da semana + badge "Hoje" na mesma linha
  if (dowEl) dowEl.innerHTML = dows[dt.getDay()];
  const day = DB.day(d);
  // Corridas automáticas do dia — busca assíncrona, cacheada por data (não
  // refaz a busca em toda chamada de renderJornada, que roda com frequência
  // inclusive a cada tick do timer da jornada ativa). Refresh periódico via
  // _syncTick (a cada 1min) cobre corridas novas capturadas durante o dia.
  if (S.autoTripsDate !== d) {
    const keepCurrent = S._autoTripsDataDate === d ? (S.autoTrips || []) : [];
    S.autoTripsDate = d;
    S.autoTrips = keepCurrent;
    loadAutoTripsForDay(d).then(list => {
      if (S.autoTripsDate !== d || !Array.isArray(list)) return; // erro mantém último dado válido
      S.autoTrips = list;
      S._autoTripsDataDate = d;
      renderJornada();
    });
  }
  day.autoTrips = S.autoTrips || [];
  // Recusadas do dia — mesmo padrão de cache do autoTrips acima. Lida aqui
  // (S.declined) e usada dentro de renderJourneyCard(), que roda logo a
  // seguir e populate o card "Total do dia".
  if (S.declinedDate !== d) {
    S.declinedDate = d;
    S.declined = [];
    loadDeclinedOffersForDay(d).then(list => {
      if (S.declinedDate !== d) return;
      S.declined = list;
      renderJornada();
    });
  }
  const rev = day.trips.reduce((s,t)=>s+(t.value||0),0) + day.revenues.reduce((s,r)=>s+(r.amount||0),0)

  if (netEl) { netEl.textContent = DB.fRI(rev-exp); netEl.className = 'v ' + (rev-exp>=0 ? 'green' : 'red'); }

  renderJourneyCard(day, isToday);
  renderMetasDoDia(day);

  // Sessões do dia (3 mais recentes, ou todas — sessão atual sempre aparece)
  const sessEl = document.getElementById('sessions-list');
  const sessToggleBtn = document.getElementById('sessions-toggle');
  if (sessEl) {
    const total = (day.sessions||[]).length;
    // Numera pela ORDEM DE HORÁRIO real (start), não pela ordem em que foi
    // salva/sincronizada no aparelho — evita "Jornada 2" mostrar um horário
    // mais tarde que "Jornada 3". origIdx continua apontando pro índice
    // verdadeiro em day.sessions, então editar/apagar continua certo.
    const byTime = (day.sessions||[]).map((s,i)=>i)
      .sort((a,b) => (day.sessions[a].start||0) - (day.sessions[b].start||0));
    const finishedAll = byTime.slice().reverse()
      .map((origIdx,ri)=>sessionItem('Jornada '+(total-ri), day.sessions[origIdx], origIdx, S.curDate, day, false));
    const finishedShown = S.sessionsExpanded ? finishedAll : finishedAll.slice(0,3);
    const liveRow = (isToday && S.gps.st==='running') ? [sessionItem('Jornada atual', {start:S.gps.start,end:null,km:S.gps.km,elapsed:getElapsed()}, -1, S.curDate, day, true)] : [];
    const rows = [...liveRow, ...finishedShown];
    sessEl.innerHTML = rows.length ? rows.join('') : '<div class="empty">Nenhuma sessão registrada</div>';
  }
  if (sessToggleBtn) {
    sessToggleBtn.style.display = (day.sessions||[]).length > 3 ? '' : 'none';
    sessToggleBtn.innerHTML = S.sessionsExpanded ? 'Ver menos' : 'Ver todas &#8250;';
  }
  // Corridas: removido daqui (23/07/2026, pedido do Yuri) — já ficam
  // acessíveis dentro do detalhe de cada jornada (seta expande e lista via
  // sessionItem, que já mescla Importadas + Automáticas). Lista de
  // Importadas ainda mora na tela Corridas separada (navTo('corridas')).
  // Lançamentos (5 mais recentes, ou todos)
  const lEl = document.getElementById('jorn-lanc-list');
  const lancToggleBtn = document.getElementById('lanc-toggle');
  if (lEl) {
    const today = DB.today();
    const allItems = [
      ...day.revenues.map(r=>Object.assign({},r,{_type:'receita'})),
      ...day.expenses.map(e=>Object.assign({},e,{_type:'despesa'}))
    ];
    // Mostrar todos os lançamentos do dia sem filtro de tab
    const items = allItems;
    const show = S.lancExpanded ? items : items.slice(0,5);
    lEl.innerHTML = show.length ? show.map(it => lancItemJornada(it)).join('') : '<div class="empty">Nenhum lançamento registrado</div>';
    if (lancToggleBtn) {
      lancToggleBtn.style.display = items.length > 5 ? '' : 'none';
      lancToggleBtn.innerHTML = S.lancExpanded ? 'Ver menos' : 'Ver todas &#8250;';
    }
  }
  reapplyOpenTripDetail();
  reapplyOpenJss();
}

function isVideoFile(f) {
  if (f.type) return f.type.startsWith('video/');
  return /\.(mp4|mov|webm|mkv|avi|m4v|3gp)$/i.test(f.name||'');
}
function handleImport(input, plat) {
  if (!input.files.length) return;
  const files = Array.from(input.files);
  closeMo('mo-import');
  const videos = files.filter(isVideoFile);
  const images = files.filter(f => !isVideoFile(f));
  if (!videos.length && !images.length) return;
  processImportBatch(images, videos, plat);
}

// Tenta de novo (até 3x) antes de desistir de um print/frame — evita que um
// erro passageiro de rede derrube a corrida inteira do import silenciosamente.
async function extractWithRetry(fn, onRetry, maxAttempts) {
  maxAttempts = maxAttempts || 3;
  let lastErr;
  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try { return await fn(); }
    catch(e) {
      lastErr = e;
      if (attempt < maxAttempts) { onRetry(attempt+1, maxAttempts); await new Promise(r => setTimeout(r, 800)); }
    }
  }
  throw lastErr;
}

// Tenta de novo (até 3x) antes de desistir de um print/frame — evita que um
// erro passageiro de rede derrube a corrida inteira do import silenciosamente.
async function extractWithRetry(fn, onRetry, maxAttempts) {
  maxAttempts = maxAttempts || 3;
  let lastErr;
  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try { return await fn(); }
    catch(e) {
      lastErr = e;
      if (attempt < maxAttempts) { onRetry(attempt+1, maxAttempts); await new Promise(r => setTimeout(r, 800)); }
    }
  }
  throw lastErr;
}
