async function loginEnter() {
  const email=document.getElementById('l-email')?.value?.trim();
  const pass =document.getElementById('l-pass')?.value;
  lsErr('l-err','');
  if (!email||!pass) { lsErr('l-err','Preencha email e senha'); return; }
  lsBtn('l-btn-enter','Entrando...',true);
  let signInResult;
  try {
    signInResult = await Promise.race([
      _SUPA.auth.signInWithPassword({ email, password: pass }),
      new Promise((_,rej) => setTimeout(() => rej(new Error('Tempo esgotado. Verifique sua conexão.')), 12000))
    ]);
  } catch(err) {
    lsBtn('l-btn-enter','Entrar',false);
    lsErr('l-err', err.message);
    return;
  }
  const { error } = signInResult;
  lsBtn('l-btn-enter','Entrar',false);
  if (error) {
    const m = error.message || '';
    let msg = 'Erro ao entrar. Tente novamente.';
    if (m.includes('Invalid login credentials') || m.includes('invalid_credentials'))
      msg = 'Email ou senha incorretos. Verifique seus dados ou use "Esqueci minha senha".';
    else if (m.includes('Email not confirmed'))
      msg = 'Email não confirmado. Use "Esqueci minha senha" para acessar.';
    else if (m.includes('Too many requests'))
      msg = 'Muitas tentativas. Aguarde alguns minutos.';
    lsErr('l-err', msg);
  }
}

async function authInit() {
  ['app-nav','app-header'].forEach(id => { const el=document.getElementById(id);if(el)el.style.display='none'; });
  document.querySelectorAll('.screen').forEach(s => s.style.display='none');
  await new Promise(r => setTimeout(r, 300));
  const { data:{ session } } = await _SUPA.auth.getSession();
  if (session?.user) { _supaUser=session.user; await loadProfile(); showApp(); }
  else showLogin();
  _SUPA.auth.onAuthStateChange(async (ev, session) => {
    if ((ev==='SIGNED_IN' || ev==='TOKEN_REFRESHED') && session?.user) {
      _supaUser = session.user;
      // Salva tokens no storage nativo (persiste mesmo com app morto)
      if (window.SmartMobiNative?.saveSession) {
        try { SmartMobiNative.saveSession(JSON.stringify({
          access_token: session.access_token,
          refresh_token: session.refresh_token
        })); } catch(e) {}
      }
      // Salva credenciais para o GpsService usar nas notificações de reporte
      // rápido e no alerta automático de dinâmico do feed. Reenviado também
      // a cada TOKEN_REFRESHED (autoRefreshToken troca o token ~1x/hora) —
      // antes só rodava no SIGNED_IN, então esses recursos paravam de
      // funcionar (silenciosamente) depois de ~1h de app aberto.
      if (window.SmartMobiNative?.saveUserToken) {
        try { SmartMobiNative.saveUserToken(session.user.id, session.access_token); } catch(e) {}
      }
      if (ev==='SIGNED_IN') { await loadProfile(); showApp(); }
    } else if (ev==='SIGNED_OUT') {
      _supaUser = null;
      if (window.SmartMobiNative?.saveSession) {
        try { SmartMobiNative.saveSession(''); } catch(e) {}
      }
      showLogin();
    }
  });
}

async function loginEnter() {
  const email=document.getElementById('l-email')?.value?.trim();
  const pass =document.getElementById('l-pass')?.value;
  lsErr('l-err','');
  if (!email||!pass) { lsErr('l-err','Preencha email e senha'); return; }
  lsBtn('l-btn-enter','Entrando...',true);
  let signInResult;
  try {
    signInResult = await Promise.race([
      _SUPA.auth.signInWithPassword({ email, password: pass }),
      new Promise((_,rej) => setTimeout(() => rej(new Error('Tempo esgotado. Verifique sua conexão.')), 12000))
    ]);
  } catch(err) {
    lsBtn('l-btn-enter','Entrar',false);
    lsErr('l-err', err.message);
    return;
  }
  const { error } = signInResult;
  lsBtn('l-btn-enter','Entrar',false);
  if (error) {
    const m = error.message || '';
    let msg = 'Erro ao entrar. Tente novamente.';
    if (m.includes('Invalid login credentials') || m.includes('invalid_credentials'))
      msg = 'Email ou senha incorretos. Verifique seus dados ou use "Esqueci minha senha".';
    else if (m.includes('Email not confirmed'))
      msg = 'Email não confirmado. Use "Esqueci minha senha" para acessar.';
    else if (m.includes('Too many requests'))
      msg = 'Muitas tentativas. Aguarde alguns minutos.';
    lsErr('l-err', msg);
  }
}

async function loadProfile() {
  if(!_supaUser) return;
  const { data } = await _SUPA.from('profiles').select('*').eq('id',_supaUser.id).single();
  if (!data) return;
  const cfg = DB.cfg();
  let changed = false;
  // Só preenche o que estiver vazio local — não sobrescreve edição que o
  // usuário já tinha feito antes deste campo existir/sincronizar direito.
  if (data.name    && !cfg.name)    { cfg.name = data.name; changed = true; }
  if (data.city     && !cfg.city)    { cfg.city = data.city; changed = true; }
  if (data.plate    && !cfg.plate)   { cfg.plate = data.plate; changed = true; }
  if (data.vehicle  && !cfg.vehicle) { cfg.vehicle = data.vehicle; changed = true; }
  // Avatar é campo novo, sem edição local legada pra proteger — nuvem manda.
  if ((data.avatar_url || null) !== (cfg.avatarUrl || null)) { cfg.avatarUrl = data.avatar_url || null; changed = true; }
  if (changed) DB.saveCfg(cfg);
}

function infHaversineM(lat1,lon1,lat2,lon2){
  const R=6371000, toRad=x=>x*Math.PI/180;
  const dLat=toRad(lat2-lat1), dLon=toRad(lon2-lon1);
  const a=Math.sin(dLat/2)**2 + Math.cos(toRad(lat1))*Math.cos(toRad(lat2))*Math.sin(dLon/2)**2;
  return R*2*Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

function feedComputeKpis(){
