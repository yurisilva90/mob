function persistGps() {

  DB.set('sm_gps_session', {

async function _persistGpsCloud() {
  if (!_supaUser || !S.gps.start || S.gps.st === 'idle') return;
  // Defesa contra instância fantasma: se outra aba/instância desta MESMA
  // origem já encerrou a jornada (apagou o marcador local), esta instância
  // deve se autodestruir em vez de continuar fazendo checkpoint de uma
  // jornada que já não existe mais oficialmente. localStorage é compartilhado
  // entre abas da mesma origem — isso funciona mesmo sem comunicação direta.
  const localMarker = DB.get('sm_gps_session');
  if (!localMarker || (localMarker.st !== 'running' && localMarker.st !== 'paused')) {
    console.warn('[_persistGpsCloud] Marcador local ausente — outra instância já encerrou. Auto-parando esta.');
    S.gps.st = 'idle'; S.gps.start = null;
    if (S.gps.wid) { try{ navigator.geolocation.clearWatch(S.gps.wid); }catch(e){} }
    if (S.gps.ti) { try{ clearInterval(S.gps.ti); }catch(e){} }
    return;
  }
  const g = S.gps;
  if (!g.sessionId) g.sessionId = DB.uid();
  const snapId = g.sessionId; // capturar antes do await
  await _SUPA.from('sessions').upsert({
    id:         snapId,
    user_id:    _supaUser.id,
    date:       _ymd(g.start), // Jornada contínua: não troca de identidade à meia-noite
    start_ms:   g.start,

    elapsed_secs: getElapsed(),
    paused_ms:  0,
    is_paused:  false,
    km_desloc:  null
  }, { onConflict: 'id' });
  // Guard de corrida: se gpsStop() foi chamado enquanto este await estava
  // pendente, o nosso write (end_ms=null) pode ter chegado ao Supabase
  // DEPOIS do write de fechamento, reabrindo a sessão. Re-aplicar o tombstone.
  if (S.gps.st === 'idle') {
    try {
      if (_gcss().has(snapId)) {
        const _cl=JSON.parse(localStorage.getItem('sm_gps_last_closed')||'null');

        await _SUPA.from('sessions').upsert({id:snapId,user_id:_supaUser.id,end_ms:_em,is_paused:false,paused_ms:0},{onConflict:'id'});
        console.log('[_persistGpsCloud] Race condition — fechamento re-aplicado:',snapId);

function _ymd(ms) { const d = new Date(ms); return d.getFullYear()+'-'+p2(d.getMonth()+1)+'-'+p2(d.getDate()); }

// Chamado a cada tick do gpsUpdate() (1x/s) e ao restaurar a sessão no boot.
// Barato o bastante pra rodar toda hora: só comparação de strings até achar
// uma virada real; o trabalho pesado (abrir modal) só acontece uma vez por
// virada, travado pelo localStorage (sm_midnight_resolved_*).
function checkMidnightCrossing() {
  const g = S.gps;
  if (g.st !== 'running' || !g.start) return;
  if (S._midnightModalOpen) return; // já perguntando, não reprocessa
  const startDate = _ymd(g.start);
  const today = DB.today();
  if (startDate === today) return; // ainda no mesmo dia — nada a fazer
  if (g.forceDate) return; // motorista já resolveu essa jornada (opção 'ontem')
  const resolvedKey = 'sm_midnight_resolved_' + (g.sessionId||'x') + '_' + today;
  if (localStorage.getItem(resolvedKey)) return; // já perguntado/resolvido pra essa virada
  openMidnightModal(startDate, today);
}

async function _flashLoadDraft() {
  if (!_bootPullDone) { showToast('Sincronizando seus dados...'); await _waitBootPull(); }
  const cfg = DB.cfg();
  const saved = cfg.flashReader;
  const defaults = getFlashDefaults();
  flashDraft = { enabled: saved && saved.enabled !== undefined ? saved.enabled : defaults.enabled, kpis: {}, recusas: {} };
  FLASH_KPI_DEFS.forEach(d => {
    const s = (saved && saved.kpis && saved.kpis[d.key]) || {};
    flashDraft.kpis[d.key] = {
      enabled: s.enabled !== undefined ? s.enabled : defaults.kpis[d.key].enabled,
      red: s.red !== undefined ? s.red : defaults.kpis[d.key].red,
      green: s.green !== undefined ? s.green : defaults.kpis[d.key].green
    };
  });
  FLASH_RECUSA_DEFS.forEach(d => {
    const s = (saved && saved.recusas && saved.recusas[d.key]) || {};
    const def = defaults.recusas[d.key];
    if (d.type === 'list') {
      flashDraft.recusas[d.key] = {
        enabled: s.enabled !== undefined ? s.enabled : def.enabled,
        // PEDIDO (17/07/2026): cada item agora é {label, type:'rua'|'bairro'}
        // pra dar pra mostrar a tag na lista — normaliza entradas antigas
        // (string solta, de antes dessa mudança) na hora de carregar.
        list: Array.isArray(s.list) ? s.list.map(_flashNormEnderecoItem).filter(Boolean) : def.list.slice()
      };
    } else {
      flashDraft.recusas[d.key] = {
        enabled: s.enabled !== undefined ? s.enabled : def.enabled,
        value: s.value !== undefined ? s.value : def.value
      };
    }
  });
}
